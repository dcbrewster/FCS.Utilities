public class CreateExcelFile
{
    private const int DEFAULT_COLUMN_WIDTH = 20;        // Set each column to be 20 wide

    public static bool CreateExcelDocument<T>(List<T> list, string xlsxFilePath)
    {
        DataSet ds = new();
        ds.Tables.Add(ListToDataTable(list));

        return CreateExcelDocument(ds, xlsxFilePath);
    }

    #region HELPER_FUNCTIONS

    public static DataTable ListToDataTable<T>(List<T> list)
    {
        DataTable dt = new();

        foreach(PropertyInfo info in typeof(T).GetProperties())
            dt.Columns.Add(new DataColumn(info.Name, GetNullableType(info.PropertyType)));

        foreach(T t in list)
        {
            DataRow row = dt.NewRow();

            foreach(PropertyInfo info in typeof(T).GetProperties())
            {
                if(!IsNullableType(info.PropertyType))
                    row[info.Name] = info.GetValue(t, null);
                else
                    row[info.Name] = (info.GetValue(t, null) ?? DBNull.Value);
            }

            dt.Rows.Add(row);
        }

        return dt;
    }

    private static Type GetNullableType(Type t)
    {
        Type returnType = t;

        if(t.IsGenericType && t.GetGenericTypeDefinition().Equals(typeof(Nullable<>)))
            returnType = Nullable.GetUnderlyingType(t);

        return returnType;
    }

    private static bool IsNullableType(Type type)
    {
        return type == typeof(string)
            || type.IsArray
            || (type.IsGenericType
                && type.GetGenericTypeDefinition().Equals(typeof(Nullable<>)));
    }

    public static bool CreateExcelDocument(DataTable dt, string xlsxFilePath)
    {
        DataSet ds = new();

        ds.Tables.Add(dt);

        bool result = CreateExcelDocument(ds, xlsxFilePath);

        ds.Tables.Remove(dt);

        return result;
    }

    #endregion HELPER_FUNCTIONS

    /// <summary>
    /// Create an Excel file, and write it to a file.
    /// </summary>
    /// <param name="ds">DataSet containing the data to be written to the Excel.</param>
    /// <param name="excelFilename">Name of file to be written.</param>
    /// <returns>True if successful, false if something went wrong.</returns>
    public static bool CreateExcelDocument(DataSet ds, string excelFilename)
    {
        try
        {
            using(SpreadsheetDocument spreadsheet = SpreadsheetDocument.Create(excelFilename, SpreadsheetDocumentType.Workbook))
            {
                WriteExcelFile(ds, spreadsheet);
            }

            Trace.WriteLine("Successfully created: " + excelFilename);

            return true;
        }
        catch(Exception ex)
        {
            Trace.WriteLine("Failed, exception thrown: " + ex.Message);

            return false;
        }
    }

    public static FileStreamResult StreamExcelDocument<T>(List<T> list, string filename)
    {
        // If you're using ASP.NET Core 2.x+, then this is the function to call !
        // It will create a temporary Excel file, stream its contents, then delete it.
        string TempFilename = Path.Join(Path.GetTempPath(), filename);

        CreateExcelDocument(list, TempFilename);

        byte[] rawData = File.ReadAllBytes(TempFilename);
        MemoryStream ms = new MemoryStream(rawData);

        FileStreamResult fr = new FileStreamResult(ms, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
        fr.FileDownloadName = filename;

        File.Delete(TempFilename);      // We don't need this anymore

        return fr;
    }

    private static void WriteExcelFile(DataSet ds, SpreadsheetDocument spreadsheet)
    {
        // Create the Excel file contents.  This function is used when creating an Excel file either writing
        // to a file, or writing to a MemoryStream.
        spreadsheet.AddWorkbookPart();
        spreadsheet.WorkbookPart.Workbook = new DocumentFormat.OpenXml.Spreadsheet.Workbook();

        DefinedNames definedNamesCol = new();

        spreadsheet.WorkbookPart.Workbook.Append(new BookViews(new WorkbookView()));

        // If we don't add a "WorkbookStylesPart", OLEDB will refuse to connect to this .xlsx file !
        WorkbookStylesPart workbookStylesPart = spreadsheet.WorkbookPart.AddNewPart<WorkbookStylesPart>("rIdStyles");
        workbookStylesPart.Stylesheet = GenerateStyleSheet();
        workbookStylesPart.Stylesheet.Save();

        // Loop through each of the DataTables in our DataSet, and create a new Excel Worksheet for each.
        uint worksheetNumber = 1;
        Sheets sheets = spreadsheet.WorkbookPart.Workbook.AppendChild(new Sheets());

        foreach(DataTable dt in ds.Tables)
        {
            // For each worksheet you want to create
            string worksheetName = dt.TableName;

            // Create worksheet part, and add it to the sheets collection in workbook
            WorksheetPart newWorksheetPart = spreadsheet.WorkbookPart.AddNewPart<WorksheetPart>();
            Sheet sheet = new()
            {
                Id = spreadsheet.WorkbookPart.GetIdOfPart(newWorksheetPart),
                SheetId = worksheetNumber,
                Name = worksheetName
            };

            // If you want to define the Column Widths for a Worksheet, you need to do this *before* appending the SheetData
            // http://social.msdn.microsoft.com/Forums/en-US/oxmlsdk/thread/1d93eca8-2949-4d12-8dd9-15cc24128b10/

            sheets.Append(sheet);

            // Append this worksheet's data to our Workbook, using OpenXmlWriter, to prevent memory problems
            WriteDataTableToExcelWorksheet(dt, newWorksheetPart, definedNamesCol);

            worksheetNumber++;
        }

        spreadsheet.WorkbookPart.Workbook.Append(definedNamesCol);
        spreadsheet.WorkbookPart.Workbook.Save();
    }

    private static Stylesheet GenerateStyleSheet()
    {
        // If you want certain Excel cells to have a different Format, color, border, fonts, etc, then you need to define a "CellFormats" records containing
        // these attributes, then assign that style number to your cell.
        //
        // For example, we'll define "Style # 3" with the attributes we'd like for our header row (Row #1) on each worksheet, where the text is a bit bigger,
        // and is white text on a dark-gray background.
        //
        // NB: The NumberFormats from 0 to 163 are hardcoded in Excel (described in the following URL), and we'll define a couple of custom number formats below.
        // https://msdn.microsoft.com/en-us/library/documentformat.openxml.spreadsheet.numberingformat.aspx
        // http://lateral8.com/articles/2010/6/11/openxml-sdk-20-formatting-excel-values.aspx
        //
        uint iExcelIndex = 164;

        return new Stylesheet(
            new NumberingFormats(
                //
                new NumberingFormat()                                                  // Custom number format # 164: especially for date-times
                {
                    NumberFormatId = UInt32Value.FromUInt32(iExcelIndex++),
                    FormatCode = StringValue.FromString("dd/MMM/yyyy hh:mm:ss")
                },
                new NumberingFormat()                                                   // Custom number format # 165: especially for date times (with a blank time)
                {
                    NumberFormatId = UInt32Value.FromUInt32(iExcelIndex++),
                    FormatCode = StringValue.FromString("dd/MMM/yyyy")
                }
           ),
            new Fonts(
                new Font(                                                               // Index 0 - The default font.
                    new FontSize() { Val = 10 },
                    new Color() { Rgb = new HexBinaryValue() { Value = "000000" } },
                    new FontName() { Val = "Arial" }),
                new Font(                                                               // Index 1 - A 12px bold font, in white.
                    new Bold(),
                    new FontSize() { Val = 12 },
                    new Color() { Rgb = new HexBinaryValue() { Value = "FFFFFF" } },
                    new FontName() { Val = "Arial" }),
                new Font(                                                               // Index 2 - An Italic font.
                    new Italic(),
                    new FontSize() { Val = 10 },
                    new Color() { Rgb = new HexBinaryValue() { Value = "000000" } },
                    new FontName() { Val = "Times New Roman" })
            ),
            new Fills(
                new Fill(                                                           // Index 0 - The default fill.
                    new PatternFill() { PatternType = PatternValues.None }),
                new Fill(                                                           // Index 1 - The default fill of gray 125 (required)
                    new PatternFill() { PatternType = PatternValues.Gray125 }),
                new Fill(                                                           // Index 2 - The yellow fill.
                    new PatternFill(
                        new ForegroundColor() { Rgb = new HexBinaryValue() { Value = "FFFFFF00" } }
                    )
                    { PatternType = PatternValues.Solid }),
                new Fill(                                                           // Index 3 - Dark-gray fill.
                    new PatternFill(
                        new ForegroundColor() { Rgb = new HexBinaryValue() { Value = "FF404040" } }
                    )
                    { PatternType = PatternValues.Solid })
            ),
            new Borders(
                new Border(                                                         // Index 0 - The default border.
                    new LeftBorder(),
                    new RightBorder(),
                    new TopBorder(),
                    new BottomBorder(),
                    new DiagonalBorder()),
                new Border(                                                         // Index 1 - Applies a Left, Right, Top, Bottom border to a cell
                    new LeftBorder(new Color() { Auto = true }) { Style = BorderStyleValues.Thin },
                    new RightBorder(new Color() { Auto = true }) { Style = BorderStyleValues.Thin },
                    new TopBorder(new Color() { Auto = true }) { Style = BorderStyleValues.Thin },
                    new BottomBorder(new Color() { Auto = true }) { Style = BorderStyleValues.Thin },
                    new DiagonalBorder())
            ),
            new CellFormats(
                new CellFormat() { FontId = 0, FillId = 0, BorderId = 0 },                         // Style # 0 - The default cell style.  If a cell does not have a style index applied it will use this style combination instead
                new CellFormat() { NumberFormatId = 164 },                                         // Style # 1 - DateTimes
                new CellFormat() { NumberFormatId = 165 },                                         // Style # 2 - Dates (with a blank time)
                new CellFormat(new Alignment() { Horizontal = HorizontalAlignmentValues.Center, Vertical = VerticalAlignmentValues.Center })
                { FontId = 1, FillId = 3, BorderId = 0, ApplyFont = true, ApplyAlignment = true },       // Style # 3 - Header row
                new CellFormat() { NumberFormatId = 3 },                                           // Style # 4 - Number format: #,##0
                new CellFormat() { NumberFormatId = 4 },                                           // Style # 5 - Number format: #,##0.00
                new CellFormat() { FontId = 1, FillId = 0, BorderId = 0, ApplyFont = true },       // Style # 6 - Bold
                new CellFormat() { FontId = 2, FillId = 0, BorderId = 0, ApplyFont = true },       // Style # 7 - Italic
                new CellFormat() { FontId = 2, FillId = 0, BorderId = 0, ApplyFont = true },       // Style # 8 - Times Roman
                new CellFormat() { FontId = 0, FillId = 2, BorderId = 0, ApplyFill = true },       // Style # 9 - Yellow Fill
                new CellFormat(                                                                    // Style # 10 - Alignment
                    new Alignment() { Horizontal = HorizontalAlignmentValues.Center, Vertical = VerticalAlignmentValues.Center }
                )
                { FontId = 0, FillId = 0, BorderId = 0, ApplyAlignment = true },
                new CellFormat() { FontId = 0, FillId = 0, BorderId = 1, ApplyBorder = true }      // Style # 11 - Border
            )
        ); // return
    }

    private static void WriteDataTableToExcelWorksheet(DataTable dt, WorksheetPart worksheetPart, DefinedNames definedNamesCol)
    {
        OpenXmlWriter writer = OpenXmlWriter.Create(worksheetPart, Encoding.ASCII);
        writer.WriteStartElement(new Worksheet());

        // To demonstrate how to set column-widths in Excel, here's how to set the width of all columns to our default of "25":
        UInt32 inx = 1;

        writer.WriteStartElement(new Columns());

        foreach(DataColumn dc in dt.Columns)
        {
            writer.WriteElement(new Column { Min = inx, Max = inx, CustomWidth = true, Width = DEFAULT_COLUMN_WIDTH });
            inx++;
        }

        writer.WriteEndElement();
        writer.WriteStartElement(new SheetData());

        string cellValue = "";
        string cellReference = "";

        // Create a Header Row in our Excel file, containing one header for each Column of data in our DataTable.
        // We'll also create an array, showing which type each column of data is (Text or Numeric), so when we come to write the actual
        // cells of data, we'll know if to write Text values or Numeric cell values.
        int numberOfColumns = dt.Columns.Count;
        bool[] IsIntegerColumn = new bool[numberOfColumns];
        bool[] IsFloatColumn = new bool[numberOfColumns];
        bool[] IsDateColumn = new bool[numberOfColumns];
        string[] excelColumnNames = new string[numberOfColumns];

        for(int n = 0; n < numberOfColumns; n++)
            excelColumnNames[n] = GetExcelColumnName(n);

        // Create the Header row in our Excel Worksheet
        // We'll set the row-height to 20px, and (using the "AppendHeaderTextCell" function) apply some formatting to the cells.
        uint rowIndex = 1;

        writer.WriteStartElement(new Row
        {
            RowIndex = rowIndex,
            Height = 20,
            CustomHeight = true
        });

        for(int colInx = 0; colInx < numberOfColumns; colInx++)
        {
            DataColumn col = dt.Columns[colInx];

            AppendHeaderTextCell(excelColumnNames[colInx] + "1", col.ColumnName, writer);
            IsIntegerColumn[colInx] = (col.DataType.FullName.StartsWith("System.Int"));
            IsFloatColumn[colInx] = (col.DataType.FullName == "System.Decimal") || (col.DataType.FullName == "System.Double") || (col.DataType.FullName == "System.Single");
            IsDateColumn[colInx] = (col.DataType.FullName == "System.DateTime");

            // Uncomment the following lines, for an example of how to create some Named Ranges in your Excel file
#if FALSE
            // For each column of data in this worksheet, let's create a Named Range, showing where there are values in this column
            //      eg  "NamedRange_UserID"  = "Drivers!$A2:$A6"
            //          "NamedRange_Surname" = "Drivers!$B2:$B6"
            string columnHeader = col.ColumnName.Replace(" ", "_");
            string NamedRange = $"{worksheetName}!${excelColumnNames[colInx]}2:${excelColumnNames[colInx]}{dt.Rows.Count + 1}";

            DefinedName definedName = new() {
                Name = "NamedRange_" + columnHeader,
                Text = NamedRange
            };

            definedNamesCol.Append(definedName);
#endif
        }
        writer.WriteEndElement();   // End of header "Row"

        // Now, step through each row of data in our DataTable...
        double cellFloatValue = 0;
        CultureInfo ci = new("en-US");

        foreach(DataRow dr in dt.Rows)
        {
            // ...create a new row, and append a set of this row's data to it.
            ++rowIndex;
            writer.WriteStartElement(new Row { RowIndex = rowIndex });

            for(int colInx = 0; colInx < numberOfColumns; colInx++)
            {
                cellValue = dr.ItemArray[colInx].ToString();
                cellValue = ReplaceHexadecimalSymbols(cellValue);
                cellReference = excelColumnNames[colInx] + rowIndex.ToString();

                // Create cell with data
                if(IsIntegerColumn[colInx] || IsFloatColumn[colInx])
                {
                    // For numeric cells without any decimal places.
                    // If this numeric value is NULL, then don't write anything to the Excel file.
                    cellFloatValue = 0;

                    bool bIncludeDecimalPlaces = IsFloatColumn[colInx];

                    if(double.TryParse(cellValue, out cellFloatValue))
                    {
                        cellValue = cellFloatValue.ToString(ci);
                        AppendNumericCell(cellReference, cellValue, bIncludeDecimalPlaces, writer);
                    }
                }
                else if(IsDateColumn[colInx])
                {
                    // For date values, we save the value to Excel as a number, but need to set the cell's style to format
                    // it as either a date or a date-time.
                    if(DateTime.TryParse(cellValue, out DateTime dateValue))
                    {
                        AppendDateCell(cellReference, dateValue, writer);
                    }
                    else
                    {
                        // This should only happen if we have a DataColumn of type "DateTime", but this particular value is null/blank.
                        AppendTextCell(cellReference, cellValue, writer);
                    }
                }
                else
                {
                    // For text cells, just write the input data straight out to the Excel file.
                    AppendTextCell(cellReference, cellValue, writer);
                }
            }

            writer.WriteEndElement(); // End of Row
        }

        writer.WriteEndElement(); // End of SheetData
        writer.WriteEndElement(); // End of worksheet
        writer.Close();
    }

    private static void AppendHeaderTextCell(string cellReference, string cellStringValue, OpenXmlWriter writer)
    {
        // Add a new "text" Cell to the first row in our Excel worksheet
        // We set these cells to use "Style # 3", so they have a gray background color & white text.
        writer.WriteElement(new Cell
        {
            CellValue = new CellValue(cellStringValue),
            CellReference = cellReference,
            DataType = CellValues.String,
            StyleIndex = 3
        });
    }

    private static void AppendTextCell(string cellReference, string cellStringValue, OpenXmlWriter writer)
    {
        // Add a new "text" Cell to our Row

#if DATA_CONTAINS_FORMULAE
        // If this item of data looks like a formula, let's store it in the Excel file as a formula rather than a string.
        if (cellStringValue.StartsWith("="))

        return $"{firstChar}{secondChar}{thirdChar}";

            Priority = Priority                             // Set Message Priority
        };

        message.From.Add(new MailboxAddress(FromName, FromAddress));

        // Add Message Recipients
        foreach (Header header in Headers) message.Headers.Add(header.Name, header.Value);                                                 // Add Any Additional Headers
        foreach (Tuple<string, string> item in ToAddressList) message.To.Add(new MailboxAddress(item.Item1, item.Item2));           // Add To Addresses
        foreach (Tuple<string, string> item in CCAddressList) message.Cc.Add(new MailboxAddress(item.Item1, item.Item2));           // Add CC Addresses
        foreach (Tuple<string, string> item in BCCAddressList) message.Bcc.Add(new MailboxAddress(item.Item1, item.Item2));         // Add BCC Addresses
        foreach (Tuple<string, string> item in ReplyToAddressList) message.ReplyTo.Add(new MailboxAddress(item.Item1, item.Item2)); // Add ReplyTo Addresses

        string? emailOverride = GetAppSetting("Email:Override");                // Check for Email Address Override

        if (emailOverride.IsValidEmailAddress())                                       // Has A Valid Override Email Address Been Supplied?
        {
            // Redirect All Email To This Address
            TextBody += BuildAddressList(message);                                     // Add All Current Recipients To End Of Message
            ClearMessageRecipients(message);                                           // Clear All Current Message Recipients
            message.To.Add(new MailboxAddress(emailOverride, emailOverride!));  // Add Override Address To Message
        }

        // Build The Message Body
        BodyBuilder builder = new()
        {
            TextBody = IsHtml ? StripHtmlFallback(TextBody) : TextBody,
            HtmlBody = IsHtml ? (string.IsNullOrWhiteSpace(HtmlBody) ? TextBody : HtmlBody) : null
        };

        foreach (string item in Attachments)
        {
            if(string.IsNullOrWhiteSpace(item)) continue;
            if(!File.Exists(item)) throw new FileNotFoundException("Email attachment was not found.", item);

            builder.Attachments.Add(item);                                      // Add File Attachments
        }

        // Add Inline Attachments
        foreach (InlineAttachment item in InlineAttachments)
        {
            if(string.IsNullOrWhiteSpace(item.FileName)) continue;
            if(!File.Exists(item.FileName)) throw new FileNotFoundException("Inline email attachment was not found.", item.FileName);

            MimeEntity image = builder.LinkedResources.Add(item.FileName);     // Add File To Message
            image.ContentId = string.IsNullOrWhiteSpace(item.CID) ? MimeUtils.GenerateMessageId() : item.CID; // Create ID For File

            //builder.linkedresources linkfile = new(item.filename)           // create linkedresource

            //    contentid = item.cid,                                          // give it an id

            //};

            //htmlView.LinkedResources.Add(LinkFile);                            // Add the LinkedResource to the appropriate view
        }

        // Add String Attachments
        foreach (StringAttachment item in StringAttachments)
        {
            byte[] content = item.ContentEncoding.GetBytes(item.Content);
            builder.Attachments.Add(item.Name, content, ContentType.Parse(item.MediaType));
        }

        message.Body = builder.ToMessageBody();                                // Add Constructed Body To Message

        return message;                                                        // Return The Message
    }

    private static string StripHtmlFallback(string value)
    {
        if(string.IsNullOrWhiteSpace(value)) return string.Empty;

        string text = value.Replace("<br />", Environment.NewLine, StringComparison.OrdinalIgnoreCase)
            .Replace("<br>", Environment.NewLine, StringComparison.OrdinalIgnoreCase);

        return WebUtility.HtmlDecode(HtmlTagRegex.Replace(text, string.Empty));
    }

    private static string BuildRedactedQueryString(IQueryCollection query)
    {
        if(query == null || query.Count == 0) return string.Empty;

        List<string> parts = [];

        foreach(KeyValuePair<string, StringValues> item in query.OrderBy(item => item.Key, StringComparer.OrdinalIgnoreCase))
        {
            if(ShouldSkipKey(item.Key)) continue;

            foreach(string? value in item.Value)
            {
                string sanitizedValue = SanitizeValue(item.Key, value ?? string.Empty);
                parts.Add($"{Uri.EscapeDataString(item.Key)}={Uri.EscapeDataString(sanitizedValue)}");
            }
        }

        return parts.Count == 0 ? string.Empty : "?" + string.Join("&", parts);
    }

/// </summary>
public static class GEO
{
    /// <summary>
    /// Calculate the distance between two lat/long points on the earth.
    /// </summary>
    public static double Distance(double lat1, double lat2, double lon1, double lon2, bool Miles = true)
    {
        lon1 = ToRadians(lon1);                         // Convert to radians
        lon2 = ToRadians(lon2);                         // Convert to radians
        lat1 = ToRadians(lat1);                         // Convert to radians
        lat2 = ToRadians(lat2);                         // Convert to radians

        // Haversine formula
        double dlon = lon2 - lon1;                      // Difference in longitude
        double dlat = lat2 - lat1;                      // Difference in latitude
        double a = Math.Pow(Math.Sin(dlat / 2), 2)
            + Math.Cos(lat1) * Math.Cos(lat2)
            * Math.Pow(Math.Sin(dlon / 2), 2);

        double c = 2 * Math.Asin(Math.Sqrt(a));

        return c * (Miles ? 3956 : 6371);                 // Return the result. Radius of earth in 6,371 kilometers or 3,956 miles.
    }

    /// Convert from Degrees to Radians
    /// </summary>
    /// <param name="angleIn10thofaDegree"></param>
    private static double ToRadians(double angleIn10thofaDegree) => (angleIn10thofaDegree * Math.PI) / 180;

/// <summary>
/// Provides utility methods for working with JSON data, including fetching JSON from URLs and posting JSON content to endpoints.
/// It also includes custom JSON converters for handling specific serialization scenarios.
/// </summary>
public static class JSON
{
    /// <summary>
    /// Get Json String
    /// </summary>
    /// <typeparam name="T">Class</typeparam>
    /// <param name="uri">Url</param>
    /// <param name="serializedOptions">Json Serializer Options</param>
    public static async Task<T[]> GetJsonFromContentAsync<T>(string uri, JsonSerializerOptions? serializedOptions = null)
    {
        HttpClient httpClient = new();                                          // Create httpClient Object
        int TryAgain = 2;                                                       // Set Control Flat

        while (TryAgain != 0)
        {
            try
            {
                HttpRequestMessage request = new(HttpMethod.Get, uri);          // Build Request

                // Send Request And Get Response
                using HttpResponseMessage response = await httpClient.SendAsync(request, HttpCompletionOption.ResponseHeadersRead).ConfigureAwait(false);

                response.EnsureSuccessStatusCode();                             // Set Flag

                if (response.IsSuccessStatusCode)                                // Check Response Code
                {
                    try
                    {
                        T[]? Result = await response.Content.ReadFromJsonAsync<T[]>(serializedOptions).ConfigureAwait(true);
                        TryAgain = 0;                                           // We're All Good

                        return Result!;
                    }
                    catch (NotSupportedException)
                    {
                        Console.WriteLine("The content type is not supported.");    // When content type is not valid
                    }
                    catch (JsonException ex)
                    {
                        // Invalid JSON
                        Console.WriteLine("Invalid JSON.");
                        Console.WriteLine($"{ex.Message}, Http {response.StatusCode}");

                        string content = string.Empty;

                        try
                        {
                            content = response.Content.ReadAsStringAsync().Result;
                        }
                        catch
                        {
                            throw new InvalidOperationException($"{ex.Message}, Http {response.StatusCode} {response.ReasonPhrase}: {content}", ex);
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                // Check For Invalid Host Error.
                if (ex.Message.Contains("No such host is known."))
                {
                    Console.WriteLine($"Host Name Is Invalid. {ex.Message.Replace("No such host is known.", "")}");
                }
                else
                {
                    Console.WriteLine($"Call To Host Just Screwed Up. Error Message:\n{ex.Message}");
                    TryAgain--;
                }
            }
        }

        return [];
    }

    /// <summary>
    /// Download JSON (string) From Url
    /// </summary>
    /// <param name="uri">URL</param>
    public static async Task<string> GetJsonAsync(string uri)
    {
        HttpClient httpClient = new();             // Create httpClient Object

        return await httpClient.GetStringAsync(uri);        // Return String From Url
    }

    /// <summary>
    /// Post JSON To Uri
    /// </summary>
    /// <param name="uri">Uri of Request</param>
    /// <param name="content">Json String</param>
    public static async Task PostJsonHttpClient(string uri, string content)
    {
        HttpResponseMessage postResponse = await new HttpClient().PostAsJsonAsync(uri, content).ConfigureAwait(false);

        postResponse.EnsureSuccessStatusCode();
    }

    /// <summary>
    /// Post JSON To Uri
    /// </summary>
    /// <param name="uri">Uri of Request</param>
    /// <param name="content">Json String</param>
    /// <returns></returns>
    private static async Task PostJsonContent(string uri, string content)
    {
        HttpClient httpClient = new();

        HttpRequestMessage postRequest = new(HttpMethod.Post, uri)
        {
            Content = JsonContent.Create(content)
        };

        HttpResponseMessage postResponse = await httpClient.SendAsync(postRequest).ConfigureAwait(false);

        postResponse.EnsureSuccessStatusCode();
    }
}

/// <summary>
/// Converts numeric JSON values to their string representation during serialization while preserving the structure of
/// arrays and objects.
/// </summary>
/// <remarks>This converter only supports write operations. The <see cref="Read"/> method throws <see
/// cref="NotImplementedException"/>. When writing, numeric values are converted to strings, and arrays and objects are
/// processed recursively to convert all nested numeric values.</remarks>
public class NumberAsStringConverter : JsonConverter<JsonElement>
{
    /// <summary>
    /// Reads and converts the JSON to a <see cref="JsonElement"/>.
    /// </summary>
    /// <param name="reader">The reader to read from.</param>
    /// <param name="typeToConvert">The type to convert.</param>
    /// <param name="options">The serializer options to use.</param>
    /// <returns>The converted <see cref="JsonElement"/>.</returns>
    /// <exception cref="NotImplementedException">This method is not implemented.</exception>
    public override JsonElement Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options) => throw new NotImplementedException();

    /// <summary>
    /// Writes the JSON element to the output, converting numeric values to string representations.
    /// </summary>
    /// <param name="writer">The writer to write to.</param>
    /// <param name="value">The JSON element to write.</param>
    /// <param name="options">The serializer options to use.</param>
    public override void Write(Utf8JsonWriter writer, JsonElement value, JsonSerializerOptions options)
    {
        switch (value.ValueKind)
        {
            case JsonValueKind.Number:
                writer.WriteStringValue(value.GetRawText());
                break;

            case JsonValueKind.Array:
                writer.WriteStartArray();

                foreach (JsonElement item in value.EnumerateArray())
                {
                    Write(writer, item, options);
                }

                writer.WriteEndArray();
                break;

            case JsonValueKind.Object:
                writer.WriteStartObject();

                foreach (JsonProperty prop in value.EnumerateObject())
                {
                    writer.WritePropertyName(prop.Name);
                    Write(writer, prop.Value, options);
                }

                writer.WriteEndObject();
                break;

            default:
                value.WriteTo(writer);
                break;
        }
    }
}

/// <summary>
/// MetaDataConverter merges the JsonPropertyName from the MetaData class with the
/// corresponding property in the original class name.
/// </summary>
public class MetaDataConverter : JsonConverter<object>
{
    /// <summary>
    /// Combine the properties from the original file with the properties of the meta file.
    /// </summary>
    /// <param name="type"></param>
    /// <returns>Combined Property List</returns>
    private static PropertyInfo[] GetCombinedProperties(Type type)
    {
        MetadataTypeAttribute? metadataAttr = type.GetCustomAttribute<MetadataTypeAttribute>();
        PropertyInfo[] metadataProps = metadataAttr?.MetadataClassType.GetProperties() ?? [];
        PropertyInfo[] originalProps = type.GetProperties();

        return [.. originalProps.Union(metadataProps)];
    }

    /// <summary>
    /// Read And Deserialize Json Data
    /// </summary>
    /// <param name="reader">JsonReader</param>
    /// <param name="typeToConvert">Property Type</param>
    /// <param name="options">Serialization Options</param>
    public override object Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    {
        using JsonDocument doc = JsonDocument.ParseValue(ref reader);           // Open The Json Document
        object result = Activator.CreateInstance(typeToConvert)!;               // Create An Object Of Requested Type
        PropertyInfo[] combinedProps = GetCombinedProperties(typeToConvert);    // Get Combined Properties From POCO And MetaData Models

        foreach (JsonProperty prop in doc.RootElement.EnumerateObject())         // Process Each Property
        {
            foreach (PropertyInfo property in combinedProps)                     // Process Each Combined Property
            {
                JsonPropertyNameAttribute? metaAttr = property.GetCustomAttribute<JsonPropertyNameAttribute>(); // Get Json Property Name From MetaData Model
                string nameToUse = metaAttr?.Name ?? property.Name;                                             // Use metaArray If Available Else Use Property Name

                if (prop.Name.Equals(nameToUse, StringComparison.OrdinalIgnoreCase))							// Did We Find Matching Property
                {
                    PropertyInfo? actualProperty = typeToConvert.GetProperty(property.Name);                    // Get The Property Info

                    object? value = null;                                                                       // Create Object

                    if (actualProperty != null)                                                                 // Do We Have A Property?
                    {
                        try
                        {
                            // Extract Property Value
                            value = JsonSerializer.Deserialize(prop.Value.GetRawText(), actualProperty.PropertyType);
                        }
                        catch (JsonException)
                        {
                            // Have String (ie 12L)
                            double newValue = 0;

                            // Calculate Value - Remove "L"
                            if (prop.Value.ToString().Length > 0) newValue = Convert.ToDouble(prop.Value.ToString()[..^1]);

                            value = JsonSerializer.Deserialize(newValue += 1000, actualProperty.PropertyType);  // Extract Property Value;
                        }

                        actualProperty.SetValue(result, value);                                                 // Set Value To Property

                        break;                                                                                  // Bail
                    }
                }
            }
        }

        return result;
    }

    /// <summary>
    /// Writes the specified value as JSON by serializing all combined properties.
    /// </summary>
    /// <param name="writer">The writer to write to.</param>
    /// <param name="value">The object to serialize.</param>
    /// <param name="options">The serializer options to use.</param>
    public override void Write(Utf8JsonWriter writer, object value, JsonSerializerOptions options)
    {
        writer.WriteStartObject();

        foreach (PropertyInfo property in GetCombinedProperties(value.GetType()))
        {
            JsonPropertyNameAttribute? metaAttr = property.GetCustomAttribute<JsonPropertyNameAttribute>();
            string? nameToUse = metaAttr?.Name ?? property.Name;
            PropertyInfo? actualProperty = value.GetType().GetProperty(property.Name);

            if (actualProperty != null)
            {
                writer.WritePropertyName(nameToUse);
                JsonSerializer.Serialize(writer, actualProperty.GetValue(value), actualProperty.PropertyType, options);
            }
        }

        writer.WriteEndObject();
    }

    /// <summary>
    /// Can this object be converted?
    /// </summary>
    /// <param name="typeToConvert">Object Type</param>
    public override bool CanConvert(Type typeToConvert) => typeof(object).IsAssignableFrom(typeToConvert);

    {

        // Use Regular Expressions To Remove HTML Tags And Replace BR Tags With Newlines
        Regex lineBreakRegex = new(@"<(br|BR)\s{0,1}\/{0,1}>", RegexOptions.Multiline | RegexOptions.ExplicitCapture, TimeSpan.FromSeconds(1));

        // This regex is designed to remove all HTML tags while preserving the content. It matches any sequence of characters that starts with '<',
        // followed by any characters that are not '>' (the tag name and attributes), and ends with '>'. The use of RegexOptions.Multiline allows
        // it to handle multi-line tags, and RegexOptions.ExplicitCapture ensures that only the entire match is captured, not any subgroups.
        Regex stripFormattingRegex = new("<[^>]*(>|$)", RegexOptions.Multiline | RegexOptions.ExplicitCapture, TimeSpan.FromSeconds(1));

        // This regex is designed to remove whitespace and line breaks between HTML tags. It matches a '>' character followed by any combination
        // of non-word characters, newlines, or carriage returns, and then a '<' character. The use of RegexOptions.Multiline allows it to handle
        // multi-line content, and RegexOptions.ExplicitCapture ensures that only the entire match is captured.
        Regex tagWhiteSpaceRegex = new(@"(>|$)(\W|\n|\r)+<", RegexOptions.Multiline | RegexOptions.ExplicitCapture, TimeSpan.FromSeconds(1));
        string text = input;

        text = Regex.Replace(text, "<style.*?</style>", "", RegexOptions.IgnoreCase | RegexOptions.Singleline, TimeSpan.FromSeconds(1));
        text = System.Net.WebUtility.HtmlDecode(text);              // Decode html specific characters
        text = tagWhiteSpaceRegex.Replace(text, "><");              // Remove tag whitespace/line breaks
        text = lineBreakRegex.Replace(text, Environment.NewLine);   // Replace <br /> with line breaks
        text = stripFormattingRegex.Replace(text, string.Empty);    // Strip formatting

    /// UseUnifiedSchedulingEngine	    false	                                                bool
    /// Volatile	                    false	                                                bool
    /// WakeToRun	                    false	                                                bool
    public static bool AddTask(string taskPath, string taskDescription, DateTime execTime, string exePath, string? workingDirectory = null)
    {
        using TaskService ts = new();                                       // Get the service on the local machine
        Task existingTask = ts.GetTask(taskPath);                           // Check if the task already exists

        if (existingTask != null) return false;                             // Task already exists - bail

        TaskDefinition? td = ts.NewTask();                                  // Create a new task definition
        td.RegistrationInfo.Description = taskDescription;                  // Add task description
        td.Principal.RunLevel = TaskRunLevel.LUA;                           // Run with limited privileges

        DateTime triggerTime = execTime;                                    // Set the trigger time to the specified execution time
        td.Triggers.Add(new DailyTrigger { StartBoundary = triggerTime });  // Create a daily trigger that starts at the specified time

        // Create an action that will launch the program whenever the trigger fires
        td.Actions.Add(new ExecAction(exePath, null, workingDirectory));

        // Register the task in the root folder
        ts.RootFolder.RegisterTaskDefinition(
            taskPath,                                                       // Task Name/Path
            td,                                                             // Task Definition
            TaskCreation.CreateOrUpdate,                                    // Create or update the task
            null,                                                           // Use current user
            null,                                                           // No password needed
            TaskLogonType.InteractiveToken,                                 // Run only when user is logged on
            null);                                                          // No SDDL-defined security descriptor needed

        td = null;                                                          // Kill Object

        return true;                                                        // Return Result
    }

namespace FCS.Utilities.DataMapping;

/// <summary>
/// Helper Class To Map DataRow Values To Entity Properties Based On Custom Attributes
/// </summary>
public static class PropertyMapHelper
{
    /// <summary>
    /// DataMapping
    /// </summary>
    /// <param name="type">Class Name</param>
    /// <param name="row">DataRow</param>
    /// <param name="prop">Property Name</param>
    /// <param name="entity">Result Entity</param>
    public static void Map([DynamicallyAccessedMembers(DynamicallyAccessedMemberTypes.PublicProperties)] Type type, DataRow row, PropertyInfo prop, object entity)
    {
        // Get DataNames For "prop" As Defined In The "type" Class
        List<string> columnNames = AttributeHelper.GetDataNames(type, prop.Name);

        // If No DataNames Defined For "prop" - Add Table Columns
        if (columnNames.Count == 0)
        {
            foreach (object c in row.Table.Columns)                     // Process Each Table Column
                columnNames.Add(c!.ToString()!);                        // Add Column Name to List
        }

        foreach (string columnName in columnNames)                      // Process Each Column
        {
            if (!string.IsNullOrWhiteSpace(columnName) && row.Table.Columns.Contains(columnName))
            {
                object propertyValue = row[columnName];

                if (propertyValue != DBNull.Value)
                {
                    ParsePrimitive(prop, entity, row[columnName]);      // Set
                    break;                                              // Bail
                }
            }
        }
    }

    /// <summary>
    /// Convert Supplied Value To Property Type And Insert In Entity
    /// </summary>
    /// <param name="prop">Property Name</param>
    /// <param name="entity">Object To Receive Value</param>
    /// <param name="value">Property Value To Insert</param>
    private static void ParsePrimitive(PropertyInfo prop, object entity, object? value)
    {
        if (prop.PropertyType == typeof(string))                                                    // Is string?
        {
            prop.SetValue(entity, value?.ToString()?.Trim(), null);
        }
        else if (prop.PropertyType == typeof(char))                                                 // Is char?
        {
            if (value == null)
            {
                prop.SetValue(entity, null, null);
            }
            else
            {
                prop.SetValue(entity, value?.ToString()?[0], null);
            }
        }
        else if (prop.PropertyType == typeof(bool) || prop.PropertyType == typeof(bool?))           // Is bool?
        {
            if (value == null)
            {
                prop.SetValue(entity, null, null);
            }
            else
            {
                prop.SetValue(entity, value?.ToString()?.StringToBoolean(), null);
            }
        }
        else if (prop.PropertyType == typeof(long))                                                 // Is long?
        {

    /// <summary>
    /// Validate the format of the postal code to ensure it meets Canada Post requirements.
    /// </summary>
    /// <param name="postalCode">Postal Code To Be Inspected</param>
    /// <returns>Validation result with formatted postal code</returns>
    public static PostalCodeValidationResult ValidateCanadianPostalCode(this string postalCode)
    {
        postalCode = postalCode.FormatCanadianPostalCode();             // Make sure postal code is in proper format

        bool isValid = CanadianPostalCodePattern().IsMatch(postalCode); // Validate the postal code against the pattern

        return new PostalCodeValidationResult(isValid, postalCode);     // Return the validation result along with the formatted postal code
    }

    /// <summary>
    /// Format a string into a Canadian Postal Code format
    /// Format: "A9A 9A9"
    /// </summary>
    /// <param name="postalCode">Postal Code To Be Inspected</param>
    public static string FormatCanadianPostalCode(this string postalCode)
    {
        if (postalCode is null) return string.Empty;

        // Make sure postal code is all upper case and remove any spaces.
        postalCode = postalCode.Trim().ToUpper().Replace(" ", "", StringComparison.Ordinal);

        // Pad the postal code to 6 characters if required
        if (postalCode.Length < 6) postalCode = postalCode.PadRight(6, '#');

        // Reformat Postal Code To "XXX XXX"
        postalCode = $"{postalCode.AsSpan()[..3]} {postalCode.AsSpan(3, 3)}";

        if (string.Equals(postalCode, "### ###", StringComparison.Ordinal)) postalCode = string.Empty;

        return postalCode;
    }

    /// <summary>
    /// Format Postal Code To All Caps But No Space
    /// </summary>
    /// <param name="postalCode">Postal Code To Be Inspected</param>
    /// <returns>Postal Code in A9A9A9 Format</returns>
    public static string FormatCanadianPostalCodeShort(this string postalCode)
    {
        string formattedPostalCode = FormatCanadianPostalCode(postalCode);

        return formattedPostalCode.Replace(" ", "", StringComparison.Ordinal);
    }

    /// <summary>
    /// Format The Postal Code For Canada, Then Check To See If It May Actually Be An

    /// Remove The "#" And Return The Remainder

    public static string FormatCanadianOrUsPostalCode(this string postalCode)

    /// <returns>Extracted section or empty string if not found</returns>

    {
        ArgumentException.ThrowIfNullOrWhiteSpace(sectionName);
        ArgumentException.ThrowIfNullOrWhiteSpace(template);

        string startName = $"<!-- {sectionName.ToUpper()} START -->";
        string endName = $"<!-- {sectionName.ToUpper()} END -->";
        int startPos = template.IndexOf(startName, StringComparison.OrdinalIgnoreCase);
        int endPos = template.IndexOf(endName, StringComparison.OrdinalIgnoreCase);

        if (startPos < 0 || endPos < 0 || startPos >= endPos) return string.Empty;

        return template.Substring(startPos, endPos - startPos + endName.Length);
    }

    public static string ExtractUrl(this string html, UrlType urlType = UrlType.All)
    {
        if (string.IsNullOrWhiteSpace(html)) return string.Empty;       // Return Empty String if Input is Null or WhiteSpace

        int srcStart = 0;                                               // Starting Position of 'src' Link
        int hrefStart = 0;                                              // Starting Position of 'href' Link
        bool imagesOnly = false;                                        // Flag to Indicate if Only Image Links Should be Returned

        if (urlType == UrlType.ImagesOnly)                              // If UrlType is ImagesOnly, Set UrlType to SrcOnly to Extract 'src' Links, Then Filter for Images
        {
            urlType = UrlType.SrcOnly;                                  // Set UrlType to SrcOnly to Extract 'src' Links, Then Filter for Images
            imagesOnly = true;                                          // Set Flag to Indicate that Only Image Links Should be Returned
        }

        // Find "src" Starting Position
        if (urlType is UrlType.All or UrlType.SrcOnly) srcStart = html.IndexOf("src=", StringComparison.OrdinalIgnoreCase);

        // Find "href" Starting Position
        if (urlType is UrlType.All or UrlType.HrefOnly) hrefStart = html.IndexOf("href=", StringComparison.OrdinalIgnoreCase);

        if (urlType == UrlType.All)             // If UrlType is All, Process Which Ever Link Was Found First
        {
            if (srcStart < 0)                   // If "src" Link Was Not Found, Set srcStart to hrefStart + 1 to Ensure That hrefStart is Processed First
                srcStart = hrefStart + 1;       // If "href" Link Was Not Found, Set hrefStart to srcStart + 1 to Ensure That srcStart is Processed First
            else if (hrefStart < 0)             // If "href" Link Was Not Found, Set
                hrefStart = srcStart + 1;       // If Both "src" and "href" Links Were Not Found, Return Empty String
        }

        // Look for 'src'
        if ((srcStart > 0 && urlType == UrlType.SrcOnly) || (hrefStart > 0 && srcStart < hrefStart && urlType == UrlType.All))
        {
            srcStart += 5;                                  // Move Starting Position to the Beginning of the URL (After 'src=')
            int posEnd = html.IndexOf('"', srcStart);       // Find the Ending Position of the URL (Next Double Quote After Starting Position)

            if (posEnd < 0) return string.Empty;            // If Ending Double Quote Was Not Found, Return Empty String

            string url = html[srcStart..posEnd];            // Extract the URL from the HTML String

            if (!imagesOnly) return url;                    // If Not Filtering for Images, Return the URL

            foreach (string filter in ImageFilters)         // Loop Through Image Filters to Check if the URL Contains Any of the Image Extensions
            {
                // If the URL Contains an Image Extension, Return the URL
                if (url.Contains($".{filter}", StringComparison.OrdinalIgnoreCase)) return url;
            }
        }

        // Look for 'href'
        if ((hrefStart > 0 && urlType == UrlType.HrefOnly) || (srcStart > 0 && hrefStart < srcStart && urlType == UrlType.All))

    private static partial Regex NameCapitalizationPattern();

    /// <summary>
    /// <para>
    /// Formats a name (or any string) so that the first character is capitalized and the
    /// remainder of the string is lower case.
    /// </para>
    /// <para>Handles names with "Mc", "Mac", "O'", "D'"</para>
    /// <para>Probably others "de", "du" that should be added.</para>
    /// <para>Use as method (string.FormatName()) or function (FormatName("String"))</para>
    /// </summary>
    /// <param name="name">The name to format</param>
    /// <returns>Formatted String</returns>
    public static string? FormatName(this string? name)
    {
        if (string.IsNullOrWhiteSpace(name)) return null;

        List<string> noTouch = ["FCS"];                                                                             // These Words Must Be Returned In Lower Case.
        List<string> allCaps = [];                                                                                  // Acronyms That Contain Vowels - Return All Upper Case
        List<string> noAllCaps = ["Jr", "Rd", "St"];                                                                // Words WithOut Vowels - But Must Be Formatted
        char[] punctuation = [.. name.Where(char.IsPunctuation).Distinct()];                                        // Get All Unique Punctuation Characters In The Name
        string[] words = [.. name.ToLower(CultureInfo.CurrentCulture).Split().Select(x => x.Trim(punctuation))];    // Split Name Into Words And Remove Punctuation
        StringBuilder result = new();                                                                               // String Builder For Result

        foreach (string word in words)
        {
            if (noTouch.Contains(word, StringComparer.Ordinal))
            {
                result.Append(' ').Append(word);
            }
            else
            {
                // First Character To Upper Case
                string term = NameCapitalizationPattern().Replace(word, m => m.Value.ToUpper(CultureInfo.CurrentCulture));

                term = term.Replace("Iii", "III", StringComparison.OrdinalIgnoreCase);
                term = term.Replace("Ii", "II", StringComparison.OrdinalIgnoreCase);
                term = term.Replace("Du ", "du ", StringComparison.OrdinalIgnoreCase);
                term = term.Replace("'S", "'s", StringComparison.OrdinalIgnoreCase);

                // If (No Vowels And Not In NoAllCaps List) Or in AllCaps List
                if ((term.Any("aeiouyAEIOUY".Contains)
                    && noAllCaps.Contains(term, StringComparer.Ordinal))
                    || allCaps.Contains(term.ToUpper(CultureInfo.CurrentCulture), StringComparer.Ordinal))
                    term = term.ToUpper(CultureInfo.CurrentCulture);

                result.Append(' ').Append(term);
            }
        }

        return result.ToString().Trim();
    }

    {

    }

    /// Generate the Hash based on the supplied Password and Salt
    /// </summary>
    /// <param name="password">Password</param>

    public string GetHash(string password, string salt)
    {
        ArgumentNullException.ThrowIfNull(password);
        ArgumentNullException.ThrowIfNull(salt);

        byte[] passwordBytes = Encoding.UTF8.GetBytes(password);
        byte[] saltBytes = GetBytes(salt);
        byte[] hash = Rfc2898DeriveBytes.Pbkdf2(passwordBytes, saltBytes, ITERATIONS_COUNT, HashAlgorithmName.SHA256, SALT_SIZE);

            "SHA256" => SHA256.Create(),
            "SHA384" => SHA384.Create(),
            "SHA512" => SHA512.Create(),
            _ => SHA256.Create() // Default to SHA256

        // ToString("x2") converts byte in hexadecimal value
        return string.Concat(hash.Select(b => b.ToString("x2"))).ToUpperInvariant();

    {

        // Look For Extension Number
        int extIndex = phone.IndexOf("ext", StringComparison.OrdinalIgnoreCase);    // Check For "ext" (case-insensitive) To Separate Extension From Main Phone Number
        string extension = string.Empty;                                            // If Extension Found, Separate It From The Main Phone Number

        }

        string? strippedPhone = phone.StripPhoneNumber();                           // Strip Main Phone Number To Digits Only

                // Remove 000 Area Code?

                Options = options
            };

            Result? result = reader.Decode(qrCodeBitmap);

            return result?.Text ?? string.Empty;

            throw new IOException($"Unable to read QR code from '{imageFileName}'", ex);
        }
    }

    {

    }

    /// <summary>
    /// Generate A Random String With A Given Size.
    /// </summary>
    /// <param name="size">Length of the string to generate</param>
    /// <param name="lowerCase">True for lowercase letters, false for uppercase</param>
    /// <returns>Random alphabetic string</returns>
    public static string RandomString(int size, bool lowerCase = false)
    {
        if (size < 1) size = 1;                 // Ensure size is at least 1

        StringBuilder builder = new(size);      // Initialize StringBuilder with specified capacity
        char offset = lowerCase ? 'a' : 'A';    // Determine ASCII offset for letters
        const int lettersOffset = 26;           // Number of letters in the alphabet

        for (int i = 0; i < size; i++)          // Generate random character and append to builder
        {
            char character = (char)Random.Shared.Next(offset, offset + lettersOffset);  // Generate random letter
            builder.Append(character);          // Append character to the builder
        }

        return builder.ToString();
    }

            .Append(RandomString(2))                    // 2 uppercase letters
            .ToString();                                // Combine parts into a single string

    /// Randomize (shuffle) the content of a string.
    /// </summary>
    /// <param name="inputString">String to be shuffled</param>
    /// <returns>Shuffled version of the input string</returns>
    public static string RandomizeString(string? inputString)
    {
        if (string.IsNullOrEmpty(inputString)) return string.Empty; // Return empty string if input is null or empty

        char[] chars = inputString.ToCharArray();                   // Convert string to character array for shuffling
        int n = chars.Length;                                       // Get the length of the character array

        // Fisher-Yates shuffle algorithm
        for (int i = n - 1; i > 0; i--)                             // Iterate from the end of the array to the beginning
        {
            int j = Random.Shared.Next(i + 1);                      // Generate a random index from 0 to i

            (chars[i], chars[j]) = (chars[j], chars[i]);            // Swap the characters at indices i and j
        }

        return new string(chars);                                   // Convert the shuffled character array back to a string and return it
    }
